12/// Identifies a unit in the backup by its encoding.
13///
14/// AlephBFT passes every unit it adds to its DAG to the [`BackupWriter`],
15/// including the units it has just replayed from this backup. Storing those
16/// again appends a full copy of the session's units on every restart within a
17/// session, and a guardian that keeps restarting in a stalled session then
18/// loads every copy at once at its next start. The backup therefore holds each
19/// unit once: the writer skips units it already stored, and the reader skips
20/// copies written before this rule existed.
21fn unit_digest(unit: &[u8]) -> sha256::Hash {
22    sha256::Hash::hash(unit)
23}

32    }
33}

35#[async_trait]
36impl aleph_bft::BackupReader for BackupReader {
37    async fn read(&mut self) -> std::io::Result<Vec<u8>> {
38        let mut dbtx = self.db.begin_transaction_nc().await;
39
40        let mut stored = 0_usize;
41        let mut loaded = BTreeSet::new();
42        let mut backup = Vec::new();
43
44        let mut units = dbtx.find_by_prefix(&AlephUnitsPrefix).await;
45        while let Some((_, unit)) = units.next().await {
46            stored += 1;
47
48            // Keep the first copy: it precedes every unit stored after it, so
49            // the units still appear after their parents.
50            if loaded.insert(unit_digest(&unit)) {
51                backup.extend_from_slice(&unit);
52            }
53        }
54
55        if stored != 0 {
56            info!(
57                target: LOG_CONSENSUS,
58                units_len = %stored,
59                unique_units_len = %loaded.len(),
60                "Recovering from an in-session-shutdown"
61            );
62        }
63
64        Ok(backup)
65    }
66}

68pub struct BackupWriter {
69    db: Database,
70    units_index: u64,
71    stored: BTreeSet<sha256::Hash>,
72}
73
74impl BackupWriter {
75    pub async fn new(db: Database) -> Self {
76        let mut units_index = 0;
77        let mut stored = BTreeSet::new();
78
79        {
80            let mut dbtx = db.begin_transaction_nc().await;
81            let mut units = dbtx.find_by_prefix(&AlephUnitsPrefix).await;
82            while let Some((key, unit)) = units.next().await {
83                units_index = units_index.max(key.0 + 1);
84                stored.insert(unit_digest(&unit));
85            }
86        }
87
88        Self {
89            db,
90            units_index,
91            stored,
92        }
93    }
94}
95
96#[async_trait]
97impl aleph_bft::BackupWriter for BackupWriter {
98    async fn append(&mut self, data: &[u8]) -> std::io::Result<()> {
99        if !self.stored.insert(unit_digest(data)) {
100            return Ok(());
101        }
102
103        let mut dbtx = self.db.begin_transaction().await;
104
105        dbtx.insert_new_entry(&AlephUnitsKey(self.units_index), &data.to_owned())
106            .await;
107
108        self.units_index += 1;
109
110        dbtx.commit_tx_result()
111            .await
112            .expect("This is the only place where we write to this key");
113
114        Ok(())
115    }
116}
