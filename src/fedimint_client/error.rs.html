19/// A failure to read or write the client's stored root secret.
20#[derive(Debug, Error)]
21#[non_exhaustive]
22pub enum ClientSecretError {
23    /// The database already holds a secret, which is never overwritten.
24    #[error("An encoded client secret already exists and cannot be overwritten")]
25    AlreadyExists,
26
27    /// The database holds no secret.
28    #[error("No encoded client secret is present in the database")]
29    NotPresent,
30
31    /// The stored secret is not a valid encoding of the requested type.
32    #[error("The stored client secret could not be decoded")]
33    Decode(#[from] DecodeError),
34}

36/// A failure to open, join or recover a client.
37#[derive(Debug, Error)]
38#[non_exhaustive]
39pub enum ClientBuildError {
40    /// The database was never joined to a federation.
41    #[error("The client database is not initialized")]
42    DatabaseNotInitialized,
43
44    /// The database already belongs to a federation and cannot be joined
45    /// again.
46    #[error("The client database is already initialized")]
47    DatabaseAlreadyInitialized,
48
49    /// The secret does not match the one the database was created with.
50    #[error("The secret does not match the one this database was created with")]
51    SecretMismatch,
52
53    /// The federation's client config could not be downloaded.
54    #[error("Failed to download the client config")]
55    ConfigDownload(#[source] Box<ClientConfigDownloadError>),
56
57    /// The stored client config could not be decoded with the modules this
58    /// client was built with.
59    #[error("Failed to decode the client config")]
60    ConfigDecode(#[from] DecodeError),
61
62    /// A database migration failed.
63    #[error("Failed to migrate the client database")]
64    Migration(#[from] DbMigrationError),
65
66    /// A module could not prepare its recovery.
67    #[error("Module {instance_id} ({kind}) failed to prepare its recovery")]
68    ModuleRecoveryPrepare {
69        /// The kind of the module that failed.
70        kind: ModuleKind,
71        /// The instance of the module that failed.
72        instance_id: ModuleInstanceId,
73        /// The failure the module reported.
74        #[source]
75        source: ClientModuleError,
76    },
77
78    /// A module could not be initialized.
79    #[error("Module {instance_id} ({kind}) failed to initialize")]
80    ModuleInit {
81        /// The kind of the module that failed.
82        kind: ModuleKind,
83        /// The instance of the module that failed.
84        instance_id: ModuleInstanceId,
85        /// The failure the module reported.
86        #[source]
87        source: ClientModuleError,
88    },
89
90    /// The database write failed.
91    #[error("Database error")]
92    Database(#[from] DatabaseError),
93
94    /// The client handle was already shut down and cannot be restarted.
95    #[error("The client is already stopped")]
96    AlreadyStopped,
97}
98
99impl From<ClientConfigDownloadError> for ClientBuildError {
100    fn from(source: ClientConfigDownloadError) -> Self {
101        Self::ConfigDownload(Box::new(source))
102    }
103}

105/// A failure of a request to [`crate::Client::handle_global_rpc`].
106///
107/// Every variant but [`Self::UnknownMethod`] carries the error of the
108/// operation the request ran, whole: its `Display` and `source()` are that
109/// error's.
110#[derive(Debug, Error)]
111#[non_exhaustive]
112pub enum GlobalRpcError {
113    /// The request's parameters do not fit the method, or its response could
114    /// not be serialized.
115    #[error(transparent)]
116    Json(#[from] serde_json::Error),
117
118    /// The federation could not report its session count.
119    #[error(transparent)]
120    Federation(#[from] FederationError),
121
122    /// A module's recovery failed, or the client stopped, while the request
123    /// waited for all recoveries.
124    #[error(transparent)]
125    Recovery(#[from] RecoveryError),
126
127    /// The backup could not be made or uploaded.
128    #[error(transparent)]
129    Backup(#[from] BackupError),
130
131    /// The client has no method of this name.
132    #[error("Unknown method: {method}")]
133    UnknownMethod {
134        /// The method the request named.
135        method: String,
136    },
137}

139/// A failure to create, encrypt, upload or read back a client backup.
140#[derive(Debug, Error)]
141#[non_exhaustive]
142pub enum BackupError {
143    /// A module is still recovering, so its state is not backed up yet.
144    #[error("Cannot back up while a module recovery is still running")]
145    PendingRecoveries,
146
147    /// The federation could not be reached.
148    #[error("The federation could not be reached")]
149    Federation(#[source] Box<FederationError>),
150
151    /// A module failed to produce its part of the backup.
152    #[error("Module {instance_id} failed to produce its backup")]
153    Module {
154        /// The module that failed.
155        instance_id: ModuleInstanceId,
156        /// The failure the module reported.
157        #[source]
158        source: ClientModuleError,
159    },
160
161    /// The encrypted backup is larger than the federation stores.
162    #[error("The backup payload is {size} bytes, over the limit of {max}")]
163    TooLarge {
164        /// The size of the encrypted backup.
165        size: usize,
166        /// The largest payload the federation stores.
167        max: usize,
168    },
169
170    /// The backup could not be encrypted or decrypted.
171    #[error("The backup could not be encrypted or decrypted")]
172    Encryption(#[source] Box<dyn std::error::Error + Send + Sync>),
173
174    /// A downloaded backup could not be decoded.
175    #[error("The backup could not be decoded")]
176    Decode(#[from] DecodeError),
177}
178
179impl From<FederationError> for BackupError {
180    fn from(source: FederationError) -> Self {
181        Self::Federation(Box::new(source))
182    }
183}

185/// A failure to wait for a module recovery to finish.
186#[derive(Debug, Error)]
187#[non_exhaustive]
188pub enum RecoveryError {
189    /// A module's recovery gave up.
190    ///
191    /// The failure is in-memory only and is never persisted: reopening the
192    /// client retries the recovery from its last persisted progress.
193    #[error("Recovery of module {module_instance_id} failed")]
194    Failed {
195        /// The module whose recovery failed.
196        module_instance_id: ModuleInstanceId,
197        /// The failure the module reported, shared by everyone waiting on the
198        /// recovery.
199        #[source]
200        source: Arc<ClientModuleError>,
201    },
202
203    /// The client shut down before the recovery reached an outcome.
204    #[error("The client shut down before the recovery finished")]
205    ClientStopped,
206}
