39/// Check if env variable is set and not equal `0` or `false` which are common
40/// ways to disable something.
41pub fn is_env_var_set(var: &str) -> bool {
42    let Some(val) = std::env::var_os(var) else {
43        return false;
44    };
45    match val.as_encoded_bytes() {
46        b"0" | b"false" => false,
47        b"1" | b"true" => true,
48        _ => {
49            warn!(
50                target: LOG_CORE,
51                %var,
52                val = %val.to_string_lossy(),
53                "Env var value invalid is invalid and ignored, assuming `true`"
54            );
55            true
56        }
57    }
58}

60/// Check if env variable is set and not equal `0` or `false` which are common
61/// ways to disable a setting. `None` if env var not set at all, which allows
62/// handling the default value.
63pub fn is_env_var_set_opt(var: &str) -> Option<bool> {
64    let val = std::env::var_os(var)?;
65    match val.as_encoded_bytes() {
66        b"0" | b"false" => Some(false),
67        b"1" | b"true" => Some(true),
68        _ => {
69            warn!(
70                target: LOG_CORE,
71                %var,
72                val = %val.to_string_lossy(),
73                "Env var value invalid is invalid and ignored"
74            );
75            None
76        }
77    }
78}

80/// Detect unit tests in the calling crate, nextest, or devimint.
81///
82/// The macro evaluates `cfg!(test)` in the calling crate. Inside a dependency's
83/// function, `cfg!(test)` describes that dependency's build and cannot detect
84/// the consuming crate's unit tests. Code compiled without `cfg(test)` still
85/// needs the runtime signal `NEXTEST` or `FM_IN_DEVIMINT`.
86///
87/// ```
88/// use fedimint_core::envs::{FM_IN_DEVIMINT_ENV, is_env_var_set, is_running_in_test_env};
89///
90/// // Doctests compile without cfg(test), like normal applications.
91/// assert!(!cfg!(test));
92/// assert_eq!(
93///     is_running_in_test_env!(),
94///     is_env_var_set("NEXTEST") || is_env_var_set(FM_IN_DEVIMINT_ENV),
95/// );
96/// ```
97#[macro_export]
98macro_rules! is_running_in_test_env {
99    () => {
100        cfg!(test)
101            || $crate::envs::is_env_var_set("NEXTEST")
102            || $crate::envs::is_env_var_set($crate::envs::FM_IN_DEVIMINT_ENV)
103    };
104}

106/// Return the delay before polling peers again for their supported module
107/// consensus versions.
108///
109/// After a restart, the first poll can run before peers' API servers bind.
110/// Retry soon if a round fails to reach all peers, so activation need not wait
111/// for the full interval. Keep polling unreachable peers at the short interval,
112/// a few requests per minute.
113///
114/// Like [`is_running_in_test_env!`], this macro checks for unit tests in the
115/// calling crate. It evaluates its argument exactly once, including in test
116/// mode.
117///
118/// ```
119/// use std::time::Duration;
120///
121/// use fedimint_core::envs::{is_running_in_test_env, next_poll_delay};
122///
123/// assert!(!cfg!(test));
124/// let in_test_env = is_running_in_test_env!();
125/// assert_eq!(
126///     next_poll_delay!(true),
127///     Duration::from_secs(if in_test_env { 5 } else { 600 }),
128/// );
129/// assert_eq!(
130///     next_poll_delay!(false),
131///     Duration::from_secs(if in_test_env { 5 } else { 30 }),
132/// );
133/// ```
134#[macro_export]
135macro_rules! next_poll_delay {
136    ($reached_all_peers:expr $(,)?) => {{
137        let reached_all_peers = $reached_all_peers;
138        if $crate::envs::is_running_in_test_env!() {
139            ::std::time::Duration::from_secs(5)
140        } else if reached_all_peers {
141            ::std::time::Duration::from_secs(600)
142        } else {
143            ::std::time::Duration::from_secs(30)
144        }
145    }};
146}

276    pub kind: String,
277    pub url: SafeUrl,
278}

280impl BitcoinRpcConfig {
281    pub fn get_defaults_from_env_vars() -> Result<Self, EnvParseError> {
282        Self::defaults_from_lookup(&|var: &str| env::var_os(var))
283    }
284
285    /// The defaults, reading the variables through `lookup` rather than from
286    /// the process environment.
287    fn defaults_from_lookup<L>(lookup: &L) -> Result<Self, EnvParseError>
288    where
289        L: Fn(&str) -> Option<OsString>,
290    {
291        let (_, kind) = first_env_var_set(
292            BITCOIN_RPC_KIND_ENVS,
293            FM_DEFAULT_BITCOIN_RPC_KIND_ENV,
294            lookup,
295        )?;
296        let (url_var, url) =
297            first_env_var_set(BITCOIN_RPC_URL_ENVS, FM_DEFAULT_BITCOIN_RPC_URL_ENV, lookup)?;
298
299        Ok(Self {
300            kind,
301            url: url.parse().map_err(|source| EnvParseError::InvalidUrl {
302                var: url_var,
303                source,
304            })?,
305        })
306    }
307}

344/// The value of the first variable of `vars` that `lookup` finds, together
345/// with the name of that variable, warning about the deprecated ones on the
346/// way.
347///
348/// `canonical` is the variable a user should set, reported when none of them
349/// is.
350///
351/// A variable that is set but does not hold valid Unicode fails even when a
352/// lower-priority variable holds a usable value, because silently ignoring a
353/// variable the user did set would surprise them.
354fn first_env_var_set<L>(
355    vars: &[(&'static str, Option<&'static str>)],
356    canonical: &'static str,
357    lookup: &L,
358) -> Result<(&'static str, String), EnvParseError>
359where
360    L: Fn(&str) -> Option<OsString>,
361{
362    for &(var, replacement) in vars {
363        let Some(value) = lookup(var) else {
364            continue;
365        };
366        let Ok(value) = value.into_string() else {
367            return Err(EnvParseError::NotUnicode { var });
368        };
369
370        if let Some(replacement) = replacement {
371            warn!(target: LOG_CORE, "{var} is obsolete, use {replacement} instead");
372        }
373
374        return Ok((var, value));
375    }
376
377    Err(EnvParseError::NotSet { var: canonical })
378}

380/// Failure to build a configuration value from environment variables.
381#[derive(Debug, Error)]
382#[non_exhaustive]
383pub enum EnvParseError {
384    /// None of the accepted variables is set; `var` is the canonical one to
385    /// set.
386    #[error("Environment variable {var} (or one of its aliases) is not set")]
387    NotSet { var: &'static str },
388    /// The variable is set but its value is not valid Unicode. A lower-priority
389    /// alias is not consulted once a variable is set.
390    #[error("Environment variable {var} is not valid Unicode")]
391    NotUnicode { var: &'static str },
392    /// The variable is set but is not a valid URL.
393    #[error("Environment variable {var} is not a valid URL")]
394    InvalidUrl {
395        var: &'static str,
396        #[source]
397        source: url::ParseError,
398    },
399}

401pub fn parse_kv_list_from_env<K, V>(env: &str) -> BTreeMap<K, V>
402where
403    K: FromStr + cmp::Ord,
404    <K as FromStr>::Err: std::error::Error,
405    V: FromStr,
406    <V as FromStr>::Err: std::error::Error,
407{
408    let mut map = BTreeMap::new();
409    let Ok(env_value) = std::env::var(env) else {
410        return BTreeMap::new();
411    };
412    for kv in env_value.split(',') {
413        let kv = kv.trim();
414
415        if kv.is_empty() {
416            continue;
417        }
418
419        if let Some((k, v)) = kv.split_once('=') {
420            let Some(k) = K::from_str(k)
421                .inspect_err(|err| {
422                    warn!(
423                        target: LOG_CORE,
424                        err = %err.fmt_compact(),
425                        "Error parsing value"
426                    );
427                })
428                .ok()
429            else {
430                continue;
431            };
432            let Some(v) = V::from_str(v)
433                .inspect_err(|err| {
434                    warn!(
435                        target: LOG_CORE,
436                        err = %err.fmt_compact(),
437                        "Error parsing value"
438                    );
439                })
440                .ok()
441            else {
442                continue;
443            };
444
445            map.insert(k, v);
446        }
447    }
448
449    map
450}
