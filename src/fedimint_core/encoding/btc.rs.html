16/// Keeps a rust-bitcoin reader failure distinguishable as [`DecodeError::Io`].
17fn from_bitcoin_encode_error(error: bitcoin::consensus::encode::Error) -> DecodeError {
18    match error {
19        bitcoin::consensus::encode::Error::Io(io) => DecodeError::Io(io.into()),
20        other => DecodeError::from_err(other),
21    }
22}

24/// Keeps a PSBT reader failure distinguishable as [`DecodeError::Io`].
25fn from_psbt_error(error: bitcoin::psbt::Error) -> DecodeError {
26    match error {
27        bitcoin::psbt::Error::Io(io) => DecodeError::Io(io.into()),
28        bitcoin::psbt::Error::ConsensusEncoding(inner) => from_bitcoin_encode_error(inner),
29        other => DecodeError::from_err(other),
30    }
31}

33fn consensus_encode_with_buffer<T: bitcoin::consensus::Encodable, W: std::io::Write>(
34    value: &T,
35    writer: &mut W,
36) -> Result<(), std::io::Error> {
37    let mut buffered_writer = std::io::BufWriter::new(writer);
38    bitcoin::consensus::Encodable::consensus_encode(value, &mut buffered_writer)?;
39    buffered_writer
40        .into_inner()
41        .map_err(std::io::IntoInnerError::into_error)?;
42    Ok(())
43}

358/// Wrap buffering support for implementations of Read.
359/// A reader which keeps an internal buffer to avoid hitting the underlying
360/// stream directly for every read.
361///
362/// In order to avoid reading bytes past the first object, and those bytes then
363/// ending up getting dropped, this BufBitcoinReader operates in
364/// one-byte-increments.
365///
366/// This code is vendored from the `lightning` crate:
367/// <https://github.com/lightningdevkit/rust-lightning/blob/5718baaed947fcaa9c60d80cdf309040c0c68489/lightning/src/util/ser.rs#L72-L138>
368struct BufBitcoinReader<'a, R: std::io::Read> {
369    inner: &'a mut R,
370    buf: [u8; 1],
371    is_consumed: bool,
372}
373
374impl<'a, R: std::io::Read> BufBitcoinReader<'a, R> {
375    /// Creates a [`BufBitcoinReader`] which will read from the given `inner`.
376    fn new(inner: &'a mut R) -> Self {
377        BufBitcoinReader {
378            inner,
379            buf: [0; 1],
380            is_consumed: true,
381        }
382    }
383}
384
385impl<R: std::io::Read> bitcoin_io::Read for BufBitcoinReader<'_, R> {
386    #[inline]
387    fn read(&mut self, output: &mut [u8]) -> bitcoin_io::Result<usize> {
388        if output.is_empty() {
389            return Ok(0);
390        }
391        #[allow(clippy::useless_let_if_seq)]
392        let mut offset = 0;
393        if !self.is_consumed {
394            output[0] = self.buf[0];
395            self.is_consumed = true;
396            offset = 1;
397        }
398        Ok(self
399            .inner
400            .read(&mut output[offset..])
401            .map(|len| len + offset)?)
402    }
403}
404
405impl<R: std::io::Read> bitcoin_io::BufRead for BufBitcoinReader<'_, R> {
406    #[inline]
407    fn fill_buf(&mut self) -> bitcoin_io::Result<&[u8]> {
408        debug_assert!(false, "rust-bitcoin doesn't actually use this");
409        if self.is_consumed {
410            let count = self.inner.read(&mut self.buf[..])?;
411            debug_assert!(count <= 1, "read gave us a garbage length");
412
413            // upon hitting EOF, assume the byte is already consumed
414            self.is_consumed = count == 0;
415        }
416
417        if self.is_consumed {
418            Ok(&[])
419        } else {
420            Ok(&self.buf[..])
421        }
422    }
423
424    #[inline]
425    fn consume(&mut self, amount: usize) {
426        debug_assert!(false, "rust-bitcoin doesn't actually use this");
427        if amount >= 1 {
428            debug_assert_eq!(amount, 1, "Can only consume one byte");
429            debug_assert!(!self.is_consumed, "Cannot consume more than had been read");
430            self.is_consumed = true;
431        }
432    }
433}

435/// A writer counting number of bytes written to it
436///
437/// Copy&pasted from <https://github.com/SOF3/count-write> which
438/// uses Apache license (and it's a trivial amount of code, repeating
439/// on stack overflow).
440pub struct BitoinIoWriteAdapter<W> {
441    inner: W,
442}
443
444impl<W> From<W> for BitoinIoWriteAdapter<W> {
445    fn from(inner: W) -> Self {
446        Self { inner }
447    }
448}
449
450impl<W: Write> bitcoin_io::Write for BitoinIoWriteAdapter<W> {
451    fn write(&mut self, buf: &[u8]) -> bitcoin_io::Result<usize> {
452        let written = self.inner.write(buf)?;
453        Ok(written)
454    }
455
456    fn flush(&mut self) -> bitcoin_io::Result<()> {
457        self.inner.flush().map_err(bitcoin_io::Error::from)
458    }
459}

476        bytes_remaining: usize,
477        bytes_written: Vec<u8>,
478    }

480    impl FailAfter {
481        fn new(bytes_remaining: usize) -> Self {
482            Self {
483                bytes_remaining,
484                bytes_written: Vec::new(),
485            }
486        }

489    impl Write for FailAfter {
490        fn write(&mut self, buffer: &[u8]) -> std::io::Result<usize> {
491            if self.bytes_remaining == 0 {
492                return Err(Error::new(ErrorKind::BrokenPipe, "injected write failure"));
493            }
494
495            let bytes_to_write = buffer.len().min(self.bytes_remaining);
496            self.bytes_written
497                .extend_from_slice(&buffer[..bytes_to_write]);
498            self.bytes_remaining -= bytes_to_write;
499            Ok(bytes_to_write)
500        }
501
502        fn flush(&mut self) -> std::io::Result<()> {
503            panic!("encoding must not flush the caller's writer");
504        }
505    }

626    #[test_log::test]
627    fn network_roundtrip() {
628        let networks: [(bitcoin::Network, [u8; 5], [u8; 4]); 5] = [
629            (
630                bitcoin::Network::Bitcoin,
631                [0xFE, 0xD9, 0xB4, 0xBE, 0xF9],
632                [0xF9, 0xBE, 0xB4, 0xD9],
633            ),
634            (
635                bitcoin::Network::Testnet,
636                [0xFE, 0x07, 0x09, 0x11, 0x0B],
637                [0x0B, 0x11, 0x09, 0x07],
638            ),
639            (
640                bitcoin::Network::Testnet4,
641                [0xFE, 0x28, 0x3F, 0x16, 0x1C],
642                [0x1C, 0x16, 0x3F, 0x28],
643            ),
644            (
645                bitcoin::Network::Signet,
646                [0xFE, 0x40, 0xCF, 0x03, 0x0A],
647                [0x0A, 0x03, 0xCF, 0x40],
648            ),
649            (
650                bitcoin::Network::Regtest,
651                [0xFE, 0xDA, 0xB5, 0xBF, 0xFA],
652                [0xFA, 0xBF, 0xB5, 0xDA],
653            ),
654        ];
655
656        for (network, magic_legacy_bytes, magic_bytes) in networks {
657            let network_legacy_encoded =
658                NetworkLegacyEncodingWrapper(network).consensus_encode_to_vec();
659
660            let network_encoded = network.consensus_encode_to_vec();
661
662            let network_legacy_decoded = NetworkLegacyEncodingWrapper::consensus_decode_whole(
663                &network_legacy_encoded,
664                &ModuleDecoderRegistry::default(),
665            )
666            .unwrap()
667            .0;
668
669            let network_decoded = bitcoin::Network::consensus_decode_whole(
670                &network_encoded,
671                &ModuleDecoderRegistry::default(),
672            )
673            .unwrap();
674
675            assert_eq!(magic_legacy_bytes, *network_legacy_encoded);
676            assert_eq!(magic_bytes, *network_encoded);
677            assert_eq!(network, network_legacy_decoded);
678            assert_eq!(network, network_decoded);
679        }
680    }

740            txid: bitcoin::Txid::from_str(
741                "51f7ed2f23e58cc6e139e715e9ce304a1e858416edc9079dd7b74fa8d2efc09a",
742            )
743            .unwrap(),
744            vout: 0,
745        };
746        let mut encoded = outpoint.consensus_encode_to_vec();
747        encoded.truncate(encoded.len() - 1);
748
749        let err =
750            bitcoin::OutPoint::consensus_decode_whole(&encoded, &ModuleDecoderRegistry::default())
751                .expect_err("35 of 36 bytes are not a full outpoint");
752        assert!(matches!(err, DecodeError::Io(_)), "{err:?}");
753    }
