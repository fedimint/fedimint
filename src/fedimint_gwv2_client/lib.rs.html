79fn is_legacy_completion_for_circuit(
80    state: &GatewayClientStateMachinesV2,
81    circuit: IncomingCircuitKey,
82) -> bool {
83    let IncomingCircuitKey {
84        incoming_chan_id,
85        htlc_id,
86    } = circuit;
87    matches!(
88        state,
89        GatewayClientStateMachinesV2::Complete(CompleteStateMachine {
90            common,
91            ..
92        }) if common.incoming_chan_id == incoming_chan_id && common.htlc_id == htlc_id
93    )

114fn incoming_relay_plan(
115    receive_exists: bool,
116    completion_exists: bool,
117    legacy_completion_exists: bool,
118) -> IncomingRelayPlan {
119    if completion_exists || legacy_completion_exists {
120        IncomingRelayPlan::Replay
121    } else if receive_exists {
122        IncomingRelayPlan::AddCompletion
123    } else {
124        IncomingRelayPlan::CreateReceiveAndCompletion
125    }

380impl GatewayClientModuleV2 {
381    /// Starts paying the invoice of an LNv2 outgoing contract, or joins the
382    /// payment already under way for it, and waits for its outcome.
383    ///
384    /// The `Ok` value is that outcome: the preimage if the payment succeeded,
385    /// or the gateway's forfeit signature, which lets the sender reclaim the
386    /// contract, if it was cancelled.
387    ///
388    /// # Errors
389    ///
390    /// Fails with a [`GatewaySendPaymentError`] if the request is refused
391    /// before any payment starts: the contract is another gateway's, the
392    /// request's signature does not verify, the federation has not confirmed
393    /// this contract at that outpoint or cannot be asked, the invoice has no
394    /// amount or does not match the contract, or the gateway cannot price the
395    /// payment.
396    pub async fn send_payment(
397        &self,
398        payload: SendPaymentPayload,
399    ) -> Result<Result<[u8; 32], Signature>, GatewaySendPaymentError> {
400        let operation_start = now();
401
402        // The operation id is equal to the contract id which also doubles as the
403        // message signed by the gateway via the forfeit signature to forfeit
404        // the gateways claim to a contract in case of cancellation. We only create a
405        // forfeit signature after we have started the send state machine to
406        // prevent replay attacks with a previously cancelled outgoing contract
407        let operation_id = OperationId::from_encodable(&payload.contract.clone());
408
409        // Since the following checks may only fail due to client side
410        // programming error we do not have to enable cancellation and can check
411        // them before we start the state machine.
412        if payload.contract.claim_pk != self.keypair.public_key() {
413            return Err(GatewaySendPaymentError::NotOurContract);
414        }
415
416        // This prevents DOS attacks where an attacker submits a different invoice.
417        if secp256k1::SECP256K1
418            .verify_schnorr(
419                &payload.auth,
420                &Message::from_digest(*payload.invoice.consensus_hash::<sha256::Hash>().as_ref()),
421                &payload.contract.refund_pk.x_only_public_key().0,
422            )
423            .is_err()
424        {
425            return Err(GatewaySendPaymentError::InvalidAuthSignature);
426        }
427
428        // The operation id is derived from the contract, which is public in the
429        // funding transaction, and joining the operation yields its preimage. So
430        // the join belongs behind the signature above.
431        if self.client_ctx.operation_exists(operation_id).await {
432            return Ok(self.subscribe_send(operation_id).await);
433        }
434
435        // We need to check that the contract has been confirmed by the federation
436        // before we start the state machine to prevent DOS attacks.
437        let (contract_id, expiration) = self
438            .module_api
439            .outgoing_contract_expiration(payload.outpoint)
440            .await?
441            .ok_or(GatewaySendPaymentError::ContractNotConfirmed)?;
442
443        if contract_id != payload.contract.contract_id() {
444            return Err(GatewaySendPaymentError::ContractIdMismatch);
445        }
446
447        let (payment_hash, amount) = match &payload.invoice {
448            LightningInvoice::Bolt11(invoice) => (
449                invoice.payment_hash(),
450                invoice
451                    .amount_milli_satoshis()
452                    .ok_or(GatewaySendPaymentError::MissingInvoiceAmount)?,
453            ),
454        };
455
456        if PaymentImage::Hash(*payment_hash) != payload.contract.payment_image {
457            return Err(GatewaySendPaymentError::PaymentHashMismatch);
458        }
459
460        let min_contract_amount = self
461            .gateway
462            .min_contract_amount(&payload.federation_id, amount)
463            .await
464            .map_err(GatewaySendPaymentError::MinContractAmount)?;
465
466        let send_sm = GatewayClientStateMachinesV2::Send(SendStateMachine {
467            common: SendSMCommon {
468                operation_id,
469                outpoint: payload.outpoint,
470                contract: payload.contract.clone(),
471                max_delay: expiration.saturating_sub(EXPIRATION_DELTA_MINIMUM_V2),
472                min_contract_amount,
473                invoice: payload.invoice,
474                claim_keypair: self.keypair,
475            },
476            state: SendSMState::Sending,
477        });
478
479        let mut dbtx = self.client_ctx.module_db().begin_transaction().await;
480        self.client_ctx
481            .manual_operation_start_dbtx(
482                &mut dbtx.to_ref_nc(),
483                operation_id,
484                LightningCommonInit::KIND.as_str(),
485                GatewayOperationMetaV2::role(GatewayOperationRoleV2::Send),
486                vec![self.client_ctx.make_dyn_state(send_sm)],
487            )
488            .await
489            .ok();
490
491        self.client_ctx
492            .log_event(
493                &mut dbtx,
494                OutgoingPaymentStarted {
495                    operation_start,
496                    outgoing_contract: payload.contract.clone(),
497                    min_contract_amount,
498                    invoice_amount: Amount::from_msats(amount),
499                    max_delay: expiration.saturating_sub(EXPIRATION_DELTA_MINIMUM_V2),
500                },
501            )
502            .await;
503        dbtx.commit_tx().await;
504
505        Ok(self.subscribe_send(operation_id).await)
506    }

508    pub async fn subscribe_send(&self, operation_id: OperationId) -> Result<[u8; 32], Signature> {
509        let mut stream = self.notifier.subscribe(operation_id).await;
510
511        loop {
512            if let Some(GatewayClientStateMachinesV2::Send(state)) = stream.next().await {
513                match state.state {
514                    SendSMState::Sending => {}
515                    SendSMState::Claiming(claiming) => {
516                        // The preimage is proof the payment succeeded, so return it to
517                        // the sender as soon as it is available rather than waiting for
518                        // an additional ordering. The gateway's claim of the outgoing
519                        // contract has already been submitted by the send state machine
520                        // and finalizes in the background.
521                        return Ok(claiming.preimage);
522                    }
523                    SendSMState::Cancelled(cancelled) => {
524                        warn!("Outgoing lightning payment is cancelled {:?}", cancelled);
525
526                        let signature = self
527                            .keypair
528                            .sign_schnorr(state.common.contract.forfeit_message());
529
530                        assert!(state.common.contract.verify_forfeit_signature(&signature));
531
532                        return Err(signature);
533                    }
534                }
535            }
536        }
537    }

562    /// Funds the incoming contract of an intercepted LNv2 HTLC and starts the
563    /// operation that completes the HTLC's circuit.
564    ///
565    /// A contract or circuit that is already being handled is joined rather
566    /// than started twice.
567    ///
568    /// # Errors
569    ///
570    /// Fails with a [`TransactionSubmitError`] if the funding transaction or
571    /// the completion operation could not be started and no earlier attempt
572    /// had started it.
573    pub async fn relay_incoming_htlc(
574        &self,
575        payment_hash: sha256::Hash,
576        incoming_chan_id: u64,
577        htlc_id: u64,
578        contract: IncomingContract,
579        amount_msat: u64,
580    ) -> Result<(), TransactionSubmitError> {
581        let operation_start = now();
582        let receive_operation_id = OperationId::from_encodable(&contract);
583        let circuit = IncomingCircuitKey {
584            incoming_chan_id,
585            htlc_id,
586        };
587        let completion_operation_id = incoming_circuit_operation_id(receive_operation_id, circuit);
588        let receive_exists = self.client_ctx.operation_exists(receive_operation_id).await;
589        let completion_exists = self
590            .client_ctx
591            .operation_exists(completion_operation_id)
592            .await;
593        let legacy_completion_exists = self
594            .legacy_completion_exists(receive_operation_id, circuit)
595            .await;
596        let plan = incoming_relay_plan(receive_exists, completion_exists, legacy_completion_exists);
597        if plan == IncomingRelayPlan::Replay {
598            return Ok(());
599        }
600
601        let commitment = contract.commitment.clone();
602        if plan == IncomingRelayPlan::CreateReceiveAndCompletion {
603            let refund_keypair = self.keypair;
604            let client_output = ClientOutput::<LightningOutput> {
605                output: LightningOutput::V0(LightningOutputV0::Incoming(contract.clone())),
606                amounts: Amounts::new_bitcoin(contract.commitment.amount),
607            };
608            let client_output_sm = ClientOutputSM::<GatewayClientStateMachinesV2> {
609                state_machines: Arc::new(move |range: OutPointRange| {
610                    assert_eq!(range.count(), 1);
611
612                    vec![GatewayClientStateMachinesV2::Receive(ReceiveStateMachine {
613                        common: ReceiveSMCommon {
614                            operation_id: receive_operation_id,
615                            contract: contract.clone(),
616                            outpoint: range.into_iter().next().unwrap(),
617                            refund_keypair,
618                        },
619                        state: ReceiveSMState::Funding,
620                    })]
621                }),
622            };
623
624            let client_output = self.client_ctx.make_client_outputs(ClientOutputBundle::new(
625                vec![client_output],
626                vec![client_output_sm],
627            ));
628            let transaction = TransactionBuilder::new().with_outputs(client_output);
629
630            let creation_result = self
631                .client_ctx
632                .finalize_and_submit_transaction(
633                    receive_operation_id,
634                    LightningCommonInit::KIND.as_str(),
635                    |_| GatewayOperationMetaV2::role(GatewayOperationRoleV2::Receive),
636                    transaction,
637                )
638                .await;
639            if let Err(error) = creation_result {
640                let operation_exists = self.client_ctx.operation_exists(receive_operation_id).await;
641                if operation_creation_failed_permanently(true, operation_exists) {
642                    return Err(error);
643                }
644            } else {
645                let mut dbtx = self.client_ctx.module_db().begin_transaction().await;
646                self.client_ctx
647                    .log_event(
648                        &mut dbtx,
649                        IncomingPaymentStarted {
650                            operation_start,
651                            incoming_contract_commitment: commitment,
652                            invoice_amount: Amount::from_msats(amount_msat),
653                        },
654                    )
655                    .await;
656                dbtx.commit_tx().await;
657            }
658        }
659
660        let completion =
661            GatewayClientStateMachinesV2::CircuitComplete(CircuitCompleteStateMachine {
662                common: CircuitCompleteSMCommon {
663                    operation_id: completion_operation_id,
664                    receive_operation_id,
665                    payment_hash,
666                    circuit,
667                },
668                state: CompleteSMState::Pending,
669            });
670        let creation_result = self
671            .client_ctx
672            .manual_operation_start(
673                completion_operation_id,
674                LightningCommonInit::KIND.as_str(),
675                GatewayOperationMetaV2::role(GatewayOperationRoleV2::CircuitCompletion),
676                vec![self.client_ctx.make_dyn_state(completion)],
677            )
678            .await;
679        if let Err(error) = creation_result {
680            let operation_exists = self
681                .client_ctx
682                .operation_exists(completion_operation_id)
683                .await;
684            if operation_creation_failed_permanently(true, operation_exists) {
685                return Err(error);
686            }
687        }
688
689        Ok(())
690    }
691
692    /// Funds the incoming contract of a direct swap and waits for its final
693    /// state.
694    ///
695    /// A swap that was already started resumes unconditionally: its operation
696    /// is keyed on the contract, so a re-entrant call — a restarted state
697    /// machine, or a concurrent payer — joins the operation in progress
698    /// instead of funding twice. `allow_fresh_dispatch` is consulted only
699    /// when no such operation exists: callers pass `false` when a wall-clock
700    /// gate such as invoice expiry forbids starting a new swap, and receive
701    /// `Ok(None)` to signal that nothing was started.
702    ///
703    /// # Errors
704    ///
705    /// Fails with a [`TransactionSubmitError`] if the funding transaction of a
706    /// fresh swap could not be submitted.
707    pub async fn relay_direct_swap(
708        &self,
709        contract: IncomingContract,
710        amount_msat: u64,
711        allow_fresh_dispatch: bool,
712    ) -> Result<Option<FinalReceiveState>, TransactionSubmitError> {
713        let operation_start = now();
714
715        let operation_id = OperationId::from_encodable(&contract);
716
717        if self.client_ctx.operation_exists(operation_id).await {
718            return Ok(Some(self.await_receive(operation_id).await));
719        }
720
721        if !allow_fresh_dispatch {
722            return Ok(None);
723        }
724
725        let refund_keypair = self.keypair;
726
727        let client_output = ClientOutput::<LightningOutput> {
728            output: LightningOutput::V0(LightningOutputV0::Incoming(contract.clone())),
729            amounts: Amounts::new_bitcoin(contract.commitment.amount),
730        };
731        let commitment = contract.commitment.clone();
732        let client_output_sm = ClientOutputSM::<GatewayClientStateMachinesV2> {
733            state_machines: Arc::new(move |range| {
734                assert_eq!(range.count(), 1);
735
736                vec![GatewayClientStateMachinesV2::Receive(ReceiveStateMachine {
737                    common: ReceiveSMCommon {
738                        operation_id,
739                        contract: contract.clone(),
740                        outpoint: range.into_iter().next().unwrap(),
741                        refund_keypair,
742                    },
743                    state: ReceiveSMState::Funding,
744                })]
745            }),
746        };
747
748        let client_output = self.client_ctx.make_client_outputs(ClientOutputBundle::new(
749            vec![client_output],
750            vec![client_output_sm],
751        ));
752
753        let transaction = TransactionBuilder::new().with_outputs(client_output);
754
755        self.client_ctx
756            .finalize_and_submit_transaction(
757                operation_id,
758                LightningCommonInit::KIND.as_str(),
759                |_| GatewayOperationMetaV2::role(GatewayOperationRoleV2::Receive),
760                transaction,
761            )
762            .await?;
763
764        let mut dbtx = self.client_ctx.module_db().begin_transaction().await;
765        self.client_ctx
766            .log_event(
767                &mut dbtx,
768                IncomingPaymentStarted {
769                    operation_start,
770                    incoming_contract_commitment: commitment,
771                    invoice_amount: Amount::from_msats(amount_msat),
772                },
773            )
774            .await;
775        dbtx.commit_tx().await;
776
777        Ok(Some(self.await_receive(operation_id).await))
778    }
779
780    pub async fn await_receive(&self, operation_id: OperationId) -> FinalReceiveState {
781        let mut stream = self.notifier.subscribe(operation_id).await;
782
783        loop {
784            if let Some(GatewayClientStateMachinesV2::Receive(state)) = stream.next().await {
785                match state.state {
786                    ReceiveSMState::Funding => {}
787                    ReceiveSMState::Rejected(..) => return FinalReceiveState::Rejected,
788                    ReceiveSMState::Success(preimage) => {
789                        return FinalReceiveState::Success(preimage);
790                    }
791                    ReceiveSMState::Refunding(out_points) => {
792                        if self
793                            .client_ctx
794                            .await_primary_module_outputs(operation_id, out_points)
795                            .await
796                            .is_err()
797                        {
798                            return FinalReceiveState::Failure;
799                        }
800
801                        return FinalReceiveState::Refunded;
802                    }
803                    ReceiveSMState::Failure => return FinalReceiveState::Failure,
804                }
805            }
806        }
807    }
808
809    /// Waits for a legacy combined receive operation or a circuit-completion
810    /// operation to finish.
811    ///
812    /// `operation_id` must identify either a legacy operation containing
813    /// [`GatewayClientStateMachinesV2::Complete`] or a role-specific operation
814    /// containing [`GatewayClientStateMachinesV2::CircuitComplete`]. Both
815    /// successful completion and a durable permanent outcome conflict terminate
816    /// the wait.
817    pub async fn await_completion(&self, operation_id: OperationId) {
818        let mut stream = self.notifier.subscribe(operation_id).await;
819
820        loop {
821            match stream.next().await {
822                Some(GatewayClientStateMachinesV2::Complete(state)) => {
823                    if matches!(
824                        state.state,
825                        CompleteSMState::Completed | CompleteSMState::CompletionFailed(_)
826                    ) {
827                        info!(%state, "LNv2 completion state machine finished");
828                        return;
829                    }
830
831                    info!(%state, "Waiting for LNv2 completion state machine");
832                }
833                Some(GatewayClientStateMachinesV2::Receive(state)) => {
834                    info!(%state, "Waiting for LNv2 completion state machine");
835                    continue;
836                }
837                Some(GatewayClientStateMachinesV2::CircuitComplete(state)) => {
838                    if matches!(
839                        state.state,
840                        CompleteSMState::Completed | CompleteSMState::CompletionFailed(_)
841                    ) {
842                        info!(%state, "LNv2 circuit completion state machine finished");
843                        return;
844                    }
845
846                    info!(%state, "Waiting for LNv2 circuit completion state machine");
847                }
848                Some(state) => {
849                    warn!(%state, "Operation is not an LNv2 completion state machine");
850                    return;
851                }
852                None => return,
853            }
854        }
855    }
856}

858/// A refusal to pay an invoice for an LNv2 client.
859///
860/// These are the checks the gateway makes before it starts paying. A payment
861/// that starts and is later cancelled is not a failure: it is reported as the
862/// forfeit signature in the `Ok` value.
863#[derive(Debug, Error)]
864#[non_exhaustive]
865pub enum GatewaySendPaymentError {
866    /// The outgoing contract names another gateway's key, so this gateway
867    /// could never claim it.
868    #[error("The outgoing contract is keyed to another gateway")]
869    NotOurContract,
870
871    /// The request's signature over the invoice does not verify against the
872    /// contract's refund key.
873    #[error("Invalid auth signature for the invoice data")]
874    InvalidAuthSignature,
875
876    /// The federation could not be asked about the outgoing contract.
877    #[error("The gateway can not reach the federation")]
878    Federation(#[source] Box<FederationError>),
879
880    /// The federation has not confirmed the outgoing contract yet.
881    #[error("The outgoing contract has not yet been confirmed")]
882    ContractNotConfirmed,
883
884    /// The contract the federation confirmed at the request's outpoint is not
885    /// the one in the request.
886    #[error("Contract Id returned by the federation does not match contract in request")]
887    ContractIdMismatch,
888
889    /// The invoice carries no amount.
890    #[error("Invoice is missing amount")]
891    MissingInvoiceAmount,
892
893    /// The invoice's payment hash is not the one the contract is locked to.
894    #[error("The invoices payment hash does not match the contracts payment hash")]
895    PaymentHashMismatch,
896
897    /// The gateway could not work out the smallest contract amount it accepts
898    /// for this payment.
899    #[error("The minimum contract amount could not be computed")]
900    MinContractAmount(#[source] GatewayClientV2Error),
901}
902
903impl From<FederationError> for GatewaySendPaymentError {
904    fn from(source: FederationError) -> Self {
905        Self::Federation(Box::new(source))
906    }
907}

909/// An interface between module implementation and the general `Gateway`
910///
911/// To abstract away and decouple the core gateway from the modules, the
912/// interface between the is expressed as a trait. The core gateway handles
913/// LNv2 operations that require access to the database or lightning node.
914#[async_trait]
915pub trait IGatewayClientV2: Debug + Send + Sync {
916    /// Uses the gateway's Lightning node to complete a payment.
917    ///
918    /// Implementations must absorb and retry every transient node or
919    /// connectivity failure. They return `Err` only when Lightning has reached
920    /// a permanent state that makes the requested outcome impossible. The
921    /// future may block while retrying and must remain cancellation-safe.
922    /// Completion state machines persist any returned error as terminal
923    /// `CompletionFailed`.
924    async fn complete_htlc(
925        &self,
926        htlc_response: InterceptPaymentResponse,
927    ) -> Result<(), LightningRpcError>;
928
929    /// Determines if the payment can be completed using a direct swap to
930    /// another federation.
931    ///
932    /// A direct swap is determined by checking the gateway's connected
933    /// lightning node against the invoice's payee lightning node. If they
934    /// are the same, then the gateway can use another client to complete
935    /// the payment be swapping ecash instead of a payment over the
936    /// Lightning network.
937    ///
938    /// # Errors
939    ///
940    /// Fails with a [`GatewayClientV2Error`] if the invoice is payable by a
941    /// direct swap but the gateway holds no incoming contract it can use for
942    /// it. The send state machine cancels the payment on a failure.
943    async fn is_direct_swap(
944        &self,
945        invoice: &Bolt11Invoice,
946    ) -> Result<Option<(IncomingContract, ClientHandleArc)>, GatewayClientV2Error>;
947
948    /// Initiates a payment over the Lightning network.
949    async fn pay(
950        &self,
951        invoice: Bolt11Invoice,
952        max_delay: u64,
953        max_fee: Amount,
954    ) -> Result<[u8; 32], LightningRpcError>;
955
956    /// Returns whether the gateway's Lightning node has any record of an
957    /// outbound payment for `payment_hash`, whatever its state.
958    ///
959    /// The send state machine consults this when it resumes after a restart:
960    /// a payment the node already knows was dispatched before the crash and
961    /// must be resolved through [`IGatewayClientV2::pay`]'s idempotent resume
962    /// path rather than cancelled by pre-dispatch checks. A wrong `false`
963    /// forfeits a contract whose payment may still settle, so implementations
964    /// must absorb transient node failures and only answer once the node's
965    /// payment store could actually be consulted.
966    async fn outbound_payment_exists(&self, payment_hash: sha256::Hash) -> bool;
967
968    /// Computes the minimum contract amount necessary for making an outgoing
969    /// payment.
970    ///
971    /// The minimum contract amount must contain transaction fees to cover the
972    /// gateway's transaction fee and optionally additional fee to cover the
973    /// gateway's Lightning fee if the payment goes over the Lightning
974    /// network.
975    ///
976    /// # Errors
977    ///
978    /// Fails with a [`GatewayClientV2Error`] if the gateway cannot price a
979    /// payment for this federation.
980    async fn min_contract_amount(
981        &self,
982        federation_id: &FederationId,
983        amount: u64,
984    ) -> Result<Amount, GatewayClientV2Error>;
985
986    /// Check if this invoice was created using LNv1 and if the gateway is
987    /// connected to the target federation.
988    async fn is_lnv1_invoice(&self, invoice: &Bolt11Invoice) -> Option<Spanned<ClientHandleArc>>;
989
990    /// Perform a swap from an LNv2 `OutgoingContract` to an LNv1
991    /// `IncomingContract`.
992    ///
993    /// A swap that was already started resumes unconditionally;
994    /// `allow_fresh_dispatch` is consulted only when no operation for this
995    /// swap exists yet. Callers pass `false` when a wall-clock gate such as
996    /// invoice expiry forbids starting a new swap, and receive `Ok(None)` to
997    /// signal that nothing was started.
998    ///
999    /// # Errors
1000    ///
1001    /// Fails with a [`GatewayClientV2Error`] if the swap could not be started
1002    /// or followed. The send state machine cancels the payment on a failure.
1003    async fn relay_lnv1_swap(
1004        &self,
1005        client: &ClientHandleArc,
1006        invoice: &Bolt11Invoice,
1007        allow_fresh_dispatch: bool,
1008    ) -> Result<Option<FinalReceiveState>, GatewayClientV2Error>;
1009
1010    /// Claims the given payment image for `operation_id` in the gateway's
1011    /// global database, returning `true` if this operation may claim the
1012    /// outgoing contract (the image was unclaimed, or already claimed by
1013    /// this same operation) and `false` if another operation already
1014    /// claimed it.
1015    ///
1016    /// A single Lightning payment yields a single preimage, so the gateway may
1017    /// claim at most one outgoing contract per payment image. Unlike the
1018    /// per-federation module database, this spans all of the gateway's
1019    /// federations, so two clients in different federations paying the same
1020    /// invoice cannot both be claimed.
1021    async fn claim_payment_image(
1022        &self,
1023        payment_image: &PaymentImage,
1024        operation_id: OperationId,
1025    ) -> bool;
1026}

1028/// A failure reported by the gateway behind [`IGatewayClientV2`].
1029///
1030/// The trait is implemented by the gateway, not by this module, so the causes
1031/// are the gateway's own. This type carries them unchanged: its `Display` and
1032/// its `source()` are the cause's.
1033#[derive(Debug, Error)]
1034#[error(transparent)]
1035pub struct GatewayClientV2Error(Box<dyn std::error::Error + Send + Sync>);
1036
1037impl GatewayClientV2Error {
1038    /// Wraps a failure of the gateway's [`IGatewayClientV2`] implementation,
1039    /// which may be any error value or a plain message.
1040    pub fn new<E>(source: E) -> Self
1041    where
1042        E: Into<Box<dyn std::error::Error + Send + Sync>>,
1043    {
1044        Self(source.into())
1045    }
1046}
