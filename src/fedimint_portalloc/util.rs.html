11pub fn open_lock_file(root_path: &Path) -> anyhow::Result<fs::File> {
12    let path = root_path.join("lock");
13    debug!(path = %path.display(), "Opening lock file...");
14    let file = fs::OpenOptions::new()
15        .create(true)
16        .append(true)
17        .read(true)
18        .open(path.clone())?;
19    debug!(path = %path.display(), "Opened lock file");
20    Ok(file)
21}

23pub fn store_json_pretty_to_file<T>(path: &Path, val: &T) -> anyhow::Result<()>
24where
25    T: Serialize,

33fn store_to_file_with<E, F>(path: &Path, f: F) -> io::Result<Result<(), E>>
34where
35    F: Fn(&mut dyn io::Write) -> Result<(), E>,
36{
37    std::fs::create_dir_all(path.parent().expect("Not a root path"))?;
38    let tmp_path = path.with_extension("tmp");
39    let mut file = std::fs::OpenOptions::new()
40        .create(true)
41        .truncate(true)
42        .write(true)
43        .open(&tmp_path)?;
44    {
45        let mut writer = BufWriter::new(&mut file);
46        if let Err(e) = f(&mut writer) {
47            return Ok(Err(e));
48        }
49        // Propagate buffered write errors before syncing or replacing the destination.
50        writer.flush()?;
51    }
52    file.sync_data()?;
53    drop(file);
54    std::fs::rename(tmp_path, path)?;
55    Ok(Ok(()))
56}
